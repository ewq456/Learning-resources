## 第一章：计算机网络基础

### IP地址

IPv4是4字节(32位)地址，分为A/B/C/D/E五类：

| 类别 | 首字节范围 | 网络号 | 主机号 | 默认掩码 |
|:----:|:----------:|:------:|:------:|:--------:|
| A | 1~127 | 1字节 | 3字节 | 255.0.0.0 |
| B | 128~191 | 2字节 | 2字节 | 255.255.0.0 |
| C | 192~223 | 3字节 | 1字节 | 255.255.255.0 |

**子网掩码**：网络号位设为1，主机号位设为0。

**子网划分**：在主机号中再划分出子网号。例如C类地址划分4个子网，则掩码为255.255.255.192（最后8位：1100 0000）。

### 端口号

端口号唯一标识主机上的应用程序，范围0~65535：
- 0~1023：知名端口（HTTP=80, SSH=22, HTTPS=443）
- 1024~49151：注册端口
- 49152~65535：动态/私有端口

### TCP vs UDP

| 特性 | TCP | UDP |
|------|:---:|:---:|
| 连接 | 面向连接 | 无连接 |
| 可靠性 | 可靠（确认/重传）| 不可靠 |
| 顺序 | 有序 | 可能乱序 |
| 开销 | 较大 | 较小 |
| 适用场景 | HTTP/FTP/SSH | DNS/视频/语音 |

### TCP三次握手

```
Client                    Server
  | ----- SYN -------->    |   第1次：客户端请求建立连接
  | <-- SYN + ACK -----    |   第2次：服务器确认并请求
  | ----- ACK -------->    |   第3次：客户端确认（防止已失效连接）
        连接建立
```

### TCP四次挥手

```
Client                    Server
  | ----- FIN -------->    |   第1次：客户端说"我说完了"
  | <----- ACK --------    |   第2次：服务器说"知道了"
  | <----- FIN --------    |   第3次：服务器说"我也说完了"
  | ----- ACK -------->    |   第4次：客户端说"知道了"
        连接释放
```

## 第二章：Socket 基础编程

### 核心API

```c
#include <sys/socket.h>

// 创建套接字
int socket(int domain, int type, int protocol);
// domain: AF_INET(IPv4), AF_INET6(IPv6), AF_UNIX(本地)
// type: SOCK_STREAM(TCP), SOCK_DGRAM(UDP)

// 绑定地址
int bind(int sockfd, const struct sockaddr *addr, socklen_t addrlen);

// 监听（TCP服务器）
int listen(int sockfd, int backlog);

// 接受连接（TCP服务器）
int accept(int sockfd, struct sockaddr *addr, socklen_t *addrlen);

// 发起连接（TCP客户端）
int connect(int sockfd, const struct sockaddr *addr, socklen_t addrlen);

// 发送和接收
ssize_t send(int sockfd, const void *buf, size_t len, int flags);
ssize_t recv(int sockfd, void *buf, size_t len, int flags);

// UDP专用
ssize_t sendto(int sockfd, const void *buf, size_t len, int flags,
               const struct sockaddr *dest_addr, socklen_t addrlen);
ssize_t recvfrom(int sockfd, void *buf, size_t len, int flags,
                 struct sockaddr *src_addr, socklen_t *addrlen);
```

### sockaddr 结构体

```c
struct sockaddr_in {
    sa_family_t    sin_family;  // AF_INET
    in_port_t      sin_port;    // 端口（网络字节序）
    struct in_addr sin_addr;    // IP地址
};

// 地址转换
inet_addr("127.0.0.1");              // 字符串→网络字节序
inet_ntoa(addr.sin_addr);            // 网络字节序→字符串
inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);  // 新版
```

### 字节序转换

```c
htons(port);     // host to network short
htonl(value);    // host to network long
ntohs(port);     // network to host short
ntohl(value);    // network to host long
```

### TCP 服务器模型

```c
int sockSer = socket(AF_INET, SOCK_STREAM, 0);

struct sockaddr_in addrSer;
addrSer.sin_family = AF_INET;
addrSer.sin_port = htons(9090);
addrSer.sin_addr.s_addr = inet_addr("127.0.0.1");

// 设置端口复用
int yes = 1;
setsockopt(sockSer, SOL_SOCKET, SO_REUSEADDR, &yes, sizeof(int));

bind(sockSer, (struct sockaddr*)&addrSer, sizeof(addrSer));
listen(sockSer, 5);  // 等待队列最大5个

while (1) {
    int sockConn = accept(sockSer, (struct sockaddr*)&addrCli, &len);
    // 处理客户端...
    close(sockConn);
}
close(sockSer);
```

### TCP 客户端模型

```c
int sockCli = socket(AF_INET, SOCK_STREAM, 0);

struct sockaddr_in addrSer;
addrSer.sin_family = AF_INET;
addrSer.sin_port = htons(9090);
addrSer.sin_addr.s_addr = inet_addr("127.0.0.1");

connect(sockCli, (struct sockaddr*)&addrSer, sizeof(addrSer));
// 发送和接收数据...
close(sockCli);
```

### UDP 模型

UDP无需listen和accept，直接收发：

```c
// 服务器
int sockSer = socket(AF_INET, SOCK_DGRAM, 0);
bind(sockSer, (struct sockaddr*)&addrSer, sizeof(addrSer));
recvfrom(sockSer, buf, size, 0, (struct sockaddr*)&addrCli, &len);
sendto(sockSer, buf, size, 0, (struct sockaddr*)&addrCli, len);

// 客户端
int sockCli = socket(AF_INET, SOCK_DGRAM, 0);
sendto(sockCli, buf, size, 0, (struct sockaddr*)&addrSer, len);
recvfrom(sockCli, buf, size, 0, (struct sockaddr*)&addrSer, &len);
```

### 套接字选项

```c
// 设置超时（变阻塞为非阻塞）
struct timeval timeout;
timeout.tv_sec = 5;
timeout.tv_usec = 0;
setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &timeout, sizeof(timeout));

// 设置发送超时
setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &timeout, sizeof(timeout));

// 设置接收缓冲区大小
int bufsize = 1024 * 1024;
setsockopt(fd, SOL_SOCKET, SO_RCVBUF, &bufsize, sizeof(bufsize));
```

## 第三章：IO多路复用

### select()

```c
#include <sys/select.h>

int select(int nfds, fd_set *readfds, fd_set *writefds,
           fd_set *exceptfds, struct timeval *timeout);

// fd_set操作宏
FD_ZERO(&set);        // 清空集合
FD_SET(fd, &set);     // 添加fd到集合
FD_CLR(fd, &set);     // 从集合移除fd
FD_ISSET(fd, &set);   // 检查fd是否在集合中
```

**select特点**：
- 监听的文件描述符有数量限制（FD_SETSIZE=1024）
- 每次调用需要将fd_set从用户态拷贝到内核态
- 内核需要遍历所有fd检查就绪状态 O(n)
- 返回后用户需要遍历所有fd找出就绪的 O(n)

### poll()

```c
#include <poll.h>

struct pollfd {
    int   fd;       // 文件描述符
    short events;   // 监听的事件
    short revents;  // 返回的事件（内核设置）
};

int poll(struct pollfd *fds, nfds_t nfds, int timeout);

// 常用事件
POLLIN    // 数据可读
POLLOUT   // 数据可写
POLLERR   // 发生错误
POLLHUP   // 连接挂起
```

**poll vs select**：
- poll没有fd数量限制（链表结构）
- 同样需要遍历 O(n)
- 不需要每次重新设置监听集合（events和revents分离）

### epoll() ⭐

```c
#include <sys/epoll.h>

// 创建epoll实例
int epoll_create(int size);  // size参数已废弃，传任意正值

// 管理fd
int epoll_ctl(int epfd, int op, int fd, struct epoll_event *event);
// op: EPOLL_CTL_ADD / EPOLL_CTL_MOD / EPOLL_CTL_DEL

struct epoll_event {
    uint32_t     events;   // EPOLLIN/EPOLLOUT/EPOLLERR等
    epoll_data_t data;     // 用户数据（通常存放fd）
};

// 等待事件
int epoll_wait(int epfd, struct epoll_event *events,
               int maxevents, int timeout);
// 返回值：就绪的fd数量
// events数组：存放就绪fd的信息（内核直接填入）
```

**epoll核心优势**：
- **事件驱动**：fd就绪时才通知，无需轮询所有fd
- 通过`epoll_ctl`注册的fd存入内核事件表（红黑树），只拷贝一次
- `epoll_wait`只返回就绪的fd（通过就绪链表），O(1)获取
- 无fd数量限制（仅受系统内存限制）

### 三种IO复用对比

| 特性 | select | poll | epoll |
|------|:------:|:----:|:-----:|
| fd数量 | 1024限制 | 无限制 | 无限制 |
| fd拷贝 | 每次拷贝 | 每次拷贝 | 一次拷贝 |
| 就绪检测 | O(n)遍历 | O(n)遍历 | O(1)回调 |
| 触发方式 | 水平触发 | 水平触发 | 水平+边缘 |
| 适用规模 | 少量连接 | 中等连接 | 大量连接 |

### epoll完整示例

```c
// 创建服务器
int listenfd = socket(AF_INET, SOCK_STREAM, 0);
bind(listenfd, ...);
listen(listenfd, LISTEN_QUEUE);

// 创建epoll
int epfd = epoll_create(1);

// 注册监听fd
struct epoll_event ev;
ev.events = EPOLLIN;
ev.data.fd = listenfd;
epoll_ctl(epfd, EPOLL_CTL_ADD, listenfd, &ev);

struct epoll_event events[MAX_EVENTS];
char buf[BUFFER_SIZE];

while (1) {
    int nready = epoll_wait(epfd, events, MAX_EVENTS, -1);

    for (int i = 0; i < nready; i++) {
        int fd = events[i].data.fd;

        if (fd == listenfd) {
            // 新连接
            int clifd = accept(listenfd, ...);
            ev.events = EPOLLIN;
            ev.data.fd = clifd;
            epoll_ctl(epfd, EPOLL_CTL_ADD, clifd, &ev);
        } else {
            // 客户端数据
            int n = read(fd, buf, sizeof(buf));
            if (n <= 0) {
                close(fd);
                epoll_ctl(epfd, EPOLL_CTL_DEL, fd, NULL);
            } else {
                write(fd, buf, n);  // 回显
            }
        }
    }
}
```

## 第四章：并发服务器模型

### 多进程模型

每个客户端fork一个子进程处理：

```c
while (1) {
    int clifd = accept(listenfd, ...);
    pid_t pid = fork();
    if (pid == 0) {
        close(listenfd);        // 子进程不需要监听fd
        // 处理客户端通信...
        close(clifd);
        exit(0);
    }
    close(clifd);  // 父进程不需要客户端fd
}
```

**缺点**：进程创建/销毁开销大，最多fork约512个进程。

### 多线程模型

每个客户端创建一个线程：

```c
void *handle_client(void *arg) {
    int fd = *(int*)arg;
    // 处理客户端...
    close(fd);
    return NULL;
}

while (1) {
    int clifd = accept(listenfd, ...);
    pthread_t tid;
    pthread_create(&tid, NULL, handle_client, &clifd);
    pthread_detach(tid);  // 分离线程，自动回收
}
```

### 线程池模型

预先创建N个线程，任务队列调度：

```c
typedef struct {
    pthread_t *threads;    // 线程数组
    int *clifds;           // 客户端fd数组
    int *flags;            // IDEL/BUSY状态
    int size;              // 池大小
    pthread_mutex_t mutex;
    pthread_cond_t cond;
} ThreadPool;
```

**优势**：避免频繁创建/销毁线程，控制资源使用。
